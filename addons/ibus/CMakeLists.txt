cmake_minimum_required(VERSION 3.20)
project(ibus-tgskk VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(IBUS REQUIRED IMPORTED_TARGET ibus-1.0>=1.5.26 gio-unix-2.0>=2.66)
get_filename_component(TGSKK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(TGSKK_RUST_LIBRARY "${TGSKK_ROOT}/target/release/libtgskk_ffi.a" CACHE FILEPATH "Built tgskk FFI library")
set(TGSKK_ZENZ_WORKER "${CMAKE_INSTALL_FULL_LIBEXECDIR}/tgskk-zenz-worker" CACHE FILEPATH "Zenz worker")
configure_file(paths.h.in paths.h @ONLY)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0>=3.24)
add_library(tgskk-ibus-config STATIC config.cpp)
target_include_directories(tgskk-ibus-config PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(tgskk-ibus-config PUBLIC PkgConfig::IBUS)
target_compile_options(tgskk-ibus-config PRIVATE -Wall -Wextra -Wpedantic)
add_library(tgskk-ibus-settings STATIC settings.cpp)
target_link_libraries(tgskk-ibus-settings PUBLIC tgskk-ibus-config PkgConfig::GTK3)
target_compile_options(tgskk-ibus-settings PRIVATE -Wall -Wextra -Wpedantic)
add_executable(ibus-setup-tgskk settings_main.cpp)
target_link_libraries(ibus-setup-tgskk PRIVATE tgskk-ibus-settings)
configure_file(ibus-setup-tgskk.desktop.in ibus-setup-tgskk.desktop @ONLY)
install(TARGETS ibus-setup-tgskk RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ibus-setup-tgskk.desktop" DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
add_library(tgskk-ibus-engine STATIC engine.cpp)
target_include_directories(tgskk-ibus-engine PUBLIC "${TGSKK_ROOT}/crates/tgskk-ffi/include" "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(tgskk-ibus-engine PUBLIC tgskk-ibus-config PkgConfig::IBUS "${TGSKK_RUST_LIBRARY}" xkbcommon dl pthread)
target_compile_options(tgskk-ibus-engine PRIVATE -Wall -Wextra -Wpedantic)
add_executable(ibus-engine-tgskk main.cpp)
target_link_libraries(ibus-engine-tgskk PRIVATE tgskk-ibus-engine)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(ibus-engine-tgskk PRIVATE -Wl,--gc-sections)
endif()
configure_file(tgskk.xml.in tgskk.xml @ONLY)
option(TGSKK_INSTALL_GOOGLE_HELPER "Install the shared Google helper" ON)
if(TGSKK_INSTALL_GOOGLE_HELPER)
  install(PROGRAMS "${TGSKK_ROOT}/target/release/gskkserv" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR} RENAME tgskk-google-skkserv)
  install(FILES "${TGSKK_ROOT}/third_party/gskkserv/LICENSE-APACHE" "${TGSKK_ROOT}/third_party/gskkserv/UPSTREAM.md" DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/tgskk/google-ime-skkserv)
endif()

install(TARGETS ibus-engine-tgskk RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/tgskk.xml" DESTINATION ${CMAKE_INSTALL_DATADIR}/ibus/component)
if(BUILD_TESTING)
  add_executable(test-ibus tests/integration.cpp)
  target_link_libraries(test-ibus PRIVATE tgskk-ibus-engine)
  target_link_options(test-ibus PRIVATE -Wl,--wrap=tgskk_session_key)
  target_compile_definitions(test-ibus PRIVATE TGSKK_SOURCE_DIR="${TGSKK_ROOT}")
  add_test(NAME ibus-integration COMMAND dbus-run-session -- $<TARGET_FILE:test-ibus>)
  set_tests_properties(ibus-integration PROPERTIES TIMEOUT 30)
  add_executable(test-ibus-client tests/client.cpp)
  target_link_libraries(test-ibus-client PRIVATE PkgConfig::IBUS)
  add_test(NAME ibus-daemon-e2e COMMAND dbus-run-session -- bash
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/session.sh"
    $<TARGET_FILE:test-ibus-client> $<TARGET_FILE:ibus-engine-tgskk>
    "${CMAKE_CURRENT_BINARY_DIR}/tgskk.xml" "${TGSKK_ROOT}/examples/SKK-JISYO.demo.utf8")
  set_tests_properties(ibus-daemon-e2e PROPERTIES TIMEOUT 30)
endif()
if(BUILD_TESTING)
  option(TGSKK_REQUIRE_GUI_TESTS "Fail configuration when GUI tests cannot run" OFF)
  find_program(XVFB_RUN xvfb-run)
  add_executable(test-ibus-launcher tests/launcher.cpp)
  target_link_libraries(test-ibus-launcher PRIVATE PkgConfig::IBUS)
  add_test(NAME ibus-settings-launcher COMMAND test-ibus-launcher
    "${CMAKE_CURRENT_BINARY_DIR}/tgskk.xml" "${CMAKE_CURRENT_BINARY_DIR}/ibus-setup-tgskk.desktop"
    "${CMAKE_INSTALL_FULL_LIBEXECDIR}/ibus-setup-tgskk" $<TARGET_FILE:ibus-setup-tgskk>)
  if(XVFB_RUN)
  add_executable(test-ibus-settings tests/settings.cpp)
  target_link_libraries(test-ibus-settings PRIVATE tgskk-ibus-settings)
  add_test(NAME ibus-settings COMMAND ${XVFB_RUN} -a $<TARGET_FILE:test-ibus-settings>)
  set_tests_properties(ibus-settings PROPERTIES TIMEOUT 30)
  elseif(TGSKK_REQUIRE_GUI_TESTS)
    message(FATAL_ERROR "xvfb-run is required for GUI tests")
  else()
    message(WARNING "xvfb-run not found; skipping only the settings GUI test")
  endif()
endif()
