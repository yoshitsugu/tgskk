cmake_minimum_required(VERSION 3.20)
project(tgskk-windows VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Use a Windows toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
get_filename_component(TGSKK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(TGSKK_RUST_LIBRARY "" CACHE FILEPATH "Windows tgskk-ffi static library")
if(NOT EXISTS "${TGSKK_RUST_LIBRARY}")
  message(FATAL_ERROR "Build tgskk-ffi and set TGSKK_RUST_LIBRARY")
endif()
add_library(tgskk-core STATIC IMPORTED)
set_target_properties(tgskk-core PROPERTIES IMPORTED_LOCATION "${TGSKK_RUST_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${TGSKK_ROOT}/crates/tgskk-ffi/include")
set(TGSKK_WINDOWS_LIBRARIES ws2_32 userenv bcrypt ntdll advapi32 user32 kernel32 synchronization powrprof crypt32)
add_library(tgskk-tip SHARED tsf.cpp inputscope_guids.cpp tgskk-tip.def)
target_compile_definitions(tgskk-tip PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(tgskk-tip PRIVATE tgskk-core ${TGSKK_WINDOWS_LIBRARIES} ole32 oleaut32 uuid shell32)
set_target_properties(tgskk-tip PROPERTIES PREFIX "")
add_executable(test-windows-core tests/core.cpp)
target_link_libraries(test-windows-core PRIVATE tgskk-core ${TGSKK_WINDOWS_LIBRARIES})
enable_testing()
add_test(NAME windows-core COMMAND test-windows-core "${TGSKK_ROOT}/examples/SKK-JISYO.demo.utf8")

add_executable(test-windows-tsf tests/tsf.cpp)
target_compile_definitions(test-windows-tsf PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(test-windows-tsf PRIVATE ole32 oleaut32 uuid user32)
add_executable(test-windows-child tests/helper.cpp)
if(MINGW)
  target_link_options(test-windows-child PRIVATE -municode)
endif()
