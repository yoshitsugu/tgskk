cmake_minimum_required(VERSION 3.20)
project(tgskk-windows VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Use a Windows toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
get_filename_component(TGSKK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(TGSKK_RUST_LIBRARY "" CACHE FILEPATH "Windows tgskk-ffi static library")
if(NOT EXISTS "${TGSKK_RUST_LIBRARY}")
  message(FATAL_ERROR "Build tgskk-ffi and set TGSKK_RUST_LIBRARY")
endif()
add_library(tgskk-core STATIC IMPORTED)
set_target_properties(tgskk-core PROPERTIES IMPORTED_LOCATION "${TGSKK_RUST_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${TGSKK_ROOT}/crates/tgskk-ffi/include")
set(TGSKK_WINDOWS_LIBRARIES ws2_32 userenv bcrypt ntdll advapi32 user32 kernel32 synchronization powrprof crypt32)
add_library(tgskk-broker-platform STATIC broker_platform.cpp)
target_compile_definitions(tgskk-broker-platform PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
target_link_libraries(tgskk-broker-platform PUBLIC ole32 shell32 advapi32 bcrypt)
add_library(tgskk-broker-client STATIC broker_client.cpp)
target_link_libraries(tgskk-broker-client PUBLIC tgskk-broker-platform tgskk-core ${TGSKK_WINDOWS_LIBRARIES})
add_executable(tgskk-broker broker.cpp)
target_link_libraries(tgskk-broker PRIVATE tgskk-broker-platform tgskk-core ${TGSKK_WINDOWS_LIBRARIES})
add_library(tgskk-tip SHARED tsf.cpp inputscope_guids.cpp tgskk-tip.def)
target_compile_definitions(tgskk-tip PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(tgskk-tip PRIVATE tgskk-broker-client ole32 oleaut32 uuid)
set_target_properties(tgskk-tip PROPERTIES PREFIX "")
add_executable(test-windows-core tests/core.cpp)
target_link_libraries(test-windows-core PRIVATE tgskk-core ${TGSKK_WINDOWS_LIBRARIES})
enable_testing()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../common" common)
add_test(NAME windows-core COMMAND test-windows-core "${TGSKK_ROOT}/examples/SKK-JISYO.demo.utf8")

add_executable(test-windows-tsf tests/tsf.cpp inputscope_guids.cpp)
target_compile_definitions(test-windows-tsf PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(test-windows-tsf PRIVATE ole32 oleaut32 uuid user32)
add_executable(test-windows-child tests/helper.cpp)
add_executable(test-windows-broker tests/broker.cpp)
target_link_libraries(test-windows-broker PRIVATE tgskk-broker-client)
add_executable(test-windows-user tests/user.cpp)
target_link_libraries(test-windows-user PRIVATE tgskk-broker-platform userenv netapi32)

if(MINGW)
  target_link_options(test-windows-child PRIVATE -municode)
  target_link_options(tgskk-broker PRIVATE -municode)
  target_link_options(test-windows-user PRIVATE -municode)
  target_link_options(test-windows-broker PRIVATE -municode)
endif()

add_executable(test-windows-config tests/config.cpp)
target_include_directories(test-windows-config PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(test-windows-config PRIVATE tgskk-broker-platform)
add_test(NAME windows-config COMMAND test-windows-config)
add_library(tgskk-settings-window STATIC settings.cpp)
target_link_libraries(tgskk-settings-window PUBLIC tgskk-broker-platform comdlg32)
add_executable(tgskk-settings WIN32 settings_main.cpp)
target_link_libraries(tgskk-settings PRIVATE tgskk-settings-window)
if(MINGW)
  target_link_options(tgskk-settings PRIVATE -municode)
endif()
add_executable(test-windows-settings tests/settings.cpp)
target_include_directories(test-windows-settings PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(test-windows-settings PRIVATE tgskk-settings-window gdi32 user32)
add_test(NAME windows-settings COMMAND test-windows-settings)
