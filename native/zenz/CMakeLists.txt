cmake_minimum_required(VERSION 3.20)
project(tgskk-zenz LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TGSKK_LLAMA_SOURCE "" CACHE PATH "Pinned, patched llama.cpp source")
if(NOT EXISTS "${TGSKK_LLAMA_SOURCE}/include/llama.h")
  message(FATAL_ERROR "Run scripts/setup-zenz.py first")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
foreach(feature COMMON TESTS TOOLS EXAMPLES SERVER APP)
  set(LLAMA_BUILD_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${TGSKK_LLAMA_SOURCE}" llama EXCLUDE_FROM_ALL)
add_executable(tgskk-zenz-worker worker.cpp)
target_link_libraries(tgskk-zenz-worker PRIVATE llama)
target_compile_options(tgskk-zenz-worker PRIVATE -Wall -Wextra -Wpedantic)
include(GNUInstallDirs)
install(TARGETS tgskk-zenz-worker RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR})
set(TGSKK_ZENZ_MODEL_FILE "" CACHE FILEPATH "Verified Zenz GGUF model")
if(EXISTS "${TGSKK_ZENZ_MODEL_FILE}")
  install(FILES "${TGSKK_ZENZ_MODEL_FILE}" DESTINATION "${CMAKE_INSTALL_DATADIR}/tgskk" RENAME zenz-v3.2-small-Q5_K_M.gguf)
endif()
install(FILES NOTICE.md DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/tgskk/zenz")
install(FILES "${TGSKK_LLAMA_SOURCE}/LICENSE" DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/tgskk/zenz" RENAME llama-LICENSE)

include(CTest)
if(BUILD_TESTING AND EXISTS "${TGSKK_ZENZ_MODEL_FILE}" AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(Python3 3.12 COMPONENTS Interpreter REQUIRED)
  add_library(zenz-backend-probe MODULE tests/backend_probe.c)
  set_target_properties(zenz-backend-probe PROPERTIES PREFIX "lib" OUTPUT_NAME "ggml-cpu-probe")
  add_test(NAME zenz-native-worker COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_worker.py"
    "$<TARGET_FILE:tgskk-zenz-worker>" "${TGSKK_ZENZ_MODEL_FILE}" "$<TARGET_FILE:zenz-backend-probe>")
  set_tests_properties(zenz-native-worker PROPERTIES TIMEOUT 30)
endif()
